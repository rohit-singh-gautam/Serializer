cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_iostream_examples LANGUAGES CXX)
  serializer_example_dependency()
  enable_testing()
endif()
add_custom_target(serializer_iostream_examples)
foreach(example IN ITEMS stringstream string_streams file_streams fstream buffered_file base_streams custom_streambuf)
  add_subdirectory("${example}")
  add_dependencies(serializer_iostream_examples "serializer_iostream_${example}")
endforeach()
