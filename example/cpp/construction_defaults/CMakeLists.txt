cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_construction_defaults_example LANGUAGES CXX)
  find_package(Serializer 2.0 CONFIG REQUIRED)
  enable_testing()
endif()

add_executable(serializer_construction_defaults_cpp main.cpp)
serializer_generate(TARGET serializer_construction_defaults_cpp SCHEMAS model.serializer
  OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(serializer_construction_defaults_cpp PRIVATE Serializer::runtime)
if(MSVC)
  target_compile_options(serializer_construction_defaults_cpp PRIVATE /W4 /WX)
else()
  target_compile_options(serializer_construction_defaults_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_test(NAME serializer_construction_defaults_cpp COMMAND serializer_construction_defaults_cpp)
endif()
