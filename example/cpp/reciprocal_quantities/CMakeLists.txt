cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_reciprocal_quantities_example LANGUAGES CXX)
  find_package(Serializer 2.0 CONFIG REQUIRED)
  enable_testing()
endif()
add_executable(serializer_reciprocal_quantities main.cpp)
serializer_generate(TARGET serializer_reciprocal_quantities
  SCHEMAS quantities.serializer CONFIG codegen.ini
  OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(serializer_reciprocal_quantities PRIVATE /W4 /WX)
else()
  target_compile_options(serializer_reciprocal_quantities PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_test(NAME serializer_reciprocal_quantities COMMAND serializer_reciprocal_quantities)
endif()
