cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_custom_code_example LANGUAGES CXX)
  find_package(Serializer CONFIG REQUIRED)
  enable_testing()
endif()

add_executable(serializer_custom_code_example main.cpp behavior.cpp)
serializer_generate(TARGET serializer_custom_code_example SCHEMAS model.serializer
  CONFIG codegen.ini OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(serializer_custom_code_example PRIVATE /W4 /WX)
else()
  target_compile_options(serializer_custom_code_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_test(NAME serializer_custom_code_example COMMAND serializer_custom_code_example)
endif()
