cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_custom_id_examples LANGUAGES C CXX)
  serializer_example_dependency()
  enable_testing()
endif()
set(SERIALIZER_EXAMPLE_LANGUAGES "cpp;c;python" CACHE STRING "Custom-ID example language folders to build")
foreach(language IN LISTS SERIALIZER_EXAMPLE_LANGUAGES)
  if(NOT language MATCHES "^(cpp|c|java|javascript|typescript|go|csharp|rust|python|swift|kotlin)$")
    message(FATAL_ERROR "Unsupported example language: ${language}")
  endif()
  add_subdirectory("${language}")
endforeach()
