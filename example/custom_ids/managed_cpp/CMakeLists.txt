cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(custom_ids_managed_example LANGUAGES CXX)
  option(SERIALIZER_BUILD_MANAGED "Build the C++ managed runtime" ON)
  serializer_example_dependency()
  enable_testing()
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
add_executable(custom_ids_managed_example main.cpp)
serializer_example_runtime_dlls(custom_ids_managed_example)
serializer_generate(TARGET custom_ids_managed_example SCHEMAS model.serializer CONFIG model.ini)
target_link_libraries(custom_ids_managed_example PRIVATE Serializer::managed)
if(MSVC)
  target_compile_options(custom_ids_managed_example PRIVATE /W4 /WX)
else()
  target_compile_options(custom_ids_managed_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
  add_test(NAME custom_ids_managed_example COMMAND custom_ids_managed_example)
endif()
