cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_managed_collaboration_example LANGUAGES CXX)
  option(SERIALIZER_BUILD_MANAGED "Build the C++ managed runtime" ON)
  serializer_example_dependency()
  enable_testing()
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
foreach(example IN ITEMS changes locks sessions undo local_sync)
  if(example STREQUAL "changes")
    set(source main.cpp)
  elseif(example STREQUAL "locks")
    set(source locks.cpp)
  elseif(example STREQUAL "sessions")
    set(source sessions.cpp)
  elseif(example STREQUAL "undo")
    set(source undo.cpp)
  else()
    set(source local_sync.cpp)
  endif()
  set(target "managed_collaboration_${example}")
  add_executable(${target} ${source})
  serializer_example_runtime_dlls(${target})
  serializer_generate(TARGET ${target} SCHEMAS ../ledger/ledger.serializer)
  target_link_libraries(${target} PRIVATE Serializer::managed)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(SERIALIZER_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
    add_test(NAME ${target} COMMAND ${target})
  endif()
endforeach()
