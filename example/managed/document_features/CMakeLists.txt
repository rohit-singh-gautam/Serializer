cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_managed_document_features LANGUAGES CXX)
  option(SERIALIZER_BUILD_MANAGED "Build the C++ managed runtime" ON)
  serializer_example_dependency()
  enable_testing()
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
add_executable(serializer_managed_document_features main.cpp)
serializer_example_runtime_dlls(serializer_managed_document_features)
serializer_generate(TARGET serializer_managed_document_features SCHEMAS document.serializer CONFIG codegen.ini)
target_link_libraries(serializer_managed_document_features PRIVATE Serializer::managed)
if(MSVC)
  target_compile_options(serializer_managed_document_features PRIVATE /W4 /WX)
else()
  target_compile_options(serializer_managed_document_features PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
  add_test(NAME serializer_managed_document_features COMMAND serializer_managed_document_features)
endif()
