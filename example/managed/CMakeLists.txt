cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_managed_example LANGUAGES CXX)
  find_package(Serializer CONFIG REQUIRED)
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
add_subdirectory(ledger)
add_subdirectory(journal)
add_subdirectory(collaboration)

# Build the point lessons independently, or all together through this target.
add_custom_target(managed_point_examples)
foreach(lesson IN ITEMS plain_point callback access managed_edit
    manual_commit scope_commit history revert labeled_history)
  add_subdirectory("${lesson}")
endforeach()
foreach(kind IN ITEMS plain callback access edit manual scoped history revert labeled_history)
  set(point_target "managed_point_${kind}")
  add_dependencies(managed_point_examples "${point_target}")
  if(MSVC)
    target_compile_options("${point_target}" PRIVATE /W4 /WX)
  else()
    target_compile_options("${point_target}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(SERIALIZER_BUILD_TESTS)
    add_test(NAME "${point_target}" COMMAND "${point_target}")
  endif()
endforeach()
