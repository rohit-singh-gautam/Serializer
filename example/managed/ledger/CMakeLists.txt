cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_managed_example LANGUAGES CXX)
  find_package(Serializer CONFIG REQUIRED)
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
add_executable(managed_example managed_example.cpp)
serializer_generate(TARGET managed_example SCHEMAS ledger.serializer)
target_link_libraries(managed_example PRIVATE Serializer::managed)
if(MSVC)
  target_compile_options(managed_example PRIVATE /W4 /WX)
else()
  target_compile_options(managed_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS)
  add_test(NAME managed_example COMMAND managed_example)
endif()
