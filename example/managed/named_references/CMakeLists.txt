cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_named_references_example LANGUAGES CXX)
  option(SERIALIZER_BUILD_MANAGED "Build the C++ managed runtime" ON)
  if(SERIALIZER_EXAMPLE_USE_INSTALLED)
    find_package(Serializer 2.31 CONFIG REQUIRED COMPONENTS runtime)
  endif()
  serializer_example_dependency()
  enable_testing()
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
add_executable(managed_named_references_example main.cpp)
serializer_example_runtime_dlls(managed_named_references_example)
serializer_generate(TARGET managed_named_references_example SCHEMAS model.serializer
  CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/serializer.ini")
target_link_libraries(managed_named_references_example PRIVATE Serializer::managed)
if(MSVC)
  target_compile_options(managed_named_references_example PRIVATE /W4 /WX)
else()
  target_compile_options(managed_named_references_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
  add_test(NAME managed_named_references_example COMMAND managed_named_references_example)
endif()
