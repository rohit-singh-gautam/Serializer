cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_managed_journal_example LANGUAGES CXX)
  option(SERIALIZER_BUILD_MANAGED "Build the C++ managed runtime" ON)
  serializer_example_dependency()
  enable_testing()
endif()
if(NOT TARGET Serializer::managed)
  message(FATAL_ERROR "Build/install Serializer with SERIALIZER_BUILD_MANAGED=ON")
endif()
add_executable(managed_journal_example main.cpp)
serializer_example_runtime_dlls(managed_journal_example)
serializer_generate(TARGET managed_journal_example SCHEMAS ../ledger/ledger.serializer)
target_link_libraries(managed_journal_example PRIVATE Serializer::managed)
if(MSVC)
  target_compile_options(managed_journal_example PRIVATE /W4 /WX)
else()
  target_compile_options(managed_journal_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
  add_test(NAME managed_journal_example
    COMMAND managed_journal_example "${CMAKE_CURRENT_BINARY_DIR}/runs")
endif()
