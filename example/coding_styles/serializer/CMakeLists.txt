cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_style_serializer LANGUAGES CXX)
  serializer_example_dependency()
  enable_testing()
endif()
add_executable(serializer_style_serializer serializer.cpp)
serializer_example_runtime_dlls(serializer_style_serializer)
serializer_generate(TARGET serializer_style_serializer SCHEMAS account.serializer CONFIG serializer.ini
  OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(serializer_style_serializer PRIVATE /W4 /WX)
else()
  target_compile_options(serializer_style_serializer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
  add_test(NAME serializer_style_serializer COMMAND serializer_style_serializer)
endif()
