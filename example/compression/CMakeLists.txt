cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_compression_examples LANGUAGES CXX)
  option(SERIALIZER_WITH_ZSTD "Enable ZSTD compression examples" ON)
  option(SERIALIZER_WITH_LZ4 "Enable LZ4 compression examples" ON)
  option(SERIALIZER_WITH_ZLIB "Enable ZLIB compression examples" ON)
  serializer_example_dependency()
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/compression_example.cmake")
add_custom_target(serializer_compression_examples)
set(compression_examples none)
foreach(backend IN ITEMS ZSTD LZ4 ZLIB)
  serializer_compression_example_available(available "${backend}")
  if(available)
    if(backend STREQUAL "ZLIB")
      list(APPEND compression_examples gzip zlib deflate custom)
    else()
      string(TOLOWER "${backend}" example)
      list(APPEND compression_examples "${example}")
    endif()
  endif()
endforeach()
message(STATUS "Compression examples enabled: ${compression_examples}")
foreach(example IN LISTS compression_examples)
  add_subdirectory("${example}")
  add_dependencies(serializer_compression_examples "serializer_compression_${example}")
endforeach()
