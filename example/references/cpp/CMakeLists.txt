cmake_minimum_required(VERSION 4.0)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(serializer_value_reference_example LANGUAGES CXX)
  if(SERIALIZER_EXAMPLE_USE_INSTALLED)
    find_package(Serializer 2.6 CONFIG REQUIRED)
  endif()
  serializer_example_dependency()
  enable_testing()
endif()

add_executable(serializer_value_reference_example main.cpp)
serializer_generate(TARGET serializer_value_reference_example SCHEMAS ../model.serializer
  OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/message.hpp"
  CONTENT "#pragma once\n#include \"model.hpp\"\n")
if(MSVC)
  target_compile_options(serializer_value_reference_example PRIVATE /W4 /WX)
else()
  target_compile_options(serializer_value_reference_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(SERIALIZER_BUILD_TESTS OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_test(NAME serializer_value_reference_example COMMAND serializer_value_reference_example)
endif()
serializer_example_runtime_dlls(serializer_value_reference_example)
