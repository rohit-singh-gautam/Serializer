cmake_minimum_required(VERSION 4.0)
project(serializer_xml_c LANGUAGES C)
find_package(LibXml2 REQUIRED)
if(NOT XML_GENERATED_DIRECTORY)
  message(FATAL_ERROR "Run example/xml/run.py to generate schema.h first")
endif()
set(XML_EXAMPLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/main.c" CACHE FILEPATH
  "Maintained XML example or qualification source")
add_executable(serializer_xml_c "${XML_EXAMPLE_SOURCE}")
target_include_directories(serializer_xml_c PRIVATE "${XML_GENERATED_DIRECTORY}")
target_link_libraries(serializer_xml_c PRIVATE LibXml2::LibXml2)
if(MSVC)
  target_compile_options(serializer_xml_c PRIVATE /std:c11 /W4 /WX /utf-8)
  target_compile_definitions(serializer_xml_c PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(serializer_xml_c PRIVATE -std=c11 -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(serializer_xml_c PRIVATE m)
endif()
