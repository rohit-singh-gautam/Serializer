cmake_minimum_required(VERSION 4.0)
project(serializer_xml_cpp LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/serializer_example_dependency.cmake")
if(NOT DEFINED SERIALIZER_WITH_XML)
  set(SERIALIZER_WITH_XML ON)
endif()
serializer_example_dependency()
enable_testing()
foreach(profile IN ITEMS configured xsd)
  if(profile STREQUAL "configured")
    set(example_target serializer_xml_cpp)
    set(example_config codegen.ini)
    set(example_dependencies)
  else()
    set(example_target serializer_xml_cpp_xsd)
    set(example_config xsd.ini)
    set(example_dependencies DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../order.xsd")
  endif()
  add_executable(${example_target} main.cpp)
  serializer_generate(TARGET ${example_target} SCHEMAS ../model.serializer
    CONFIG "${CMAKE_CURRENT_LIST_DIR}/../${example_config}"
    ${example_dependencies}
    OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${profile}/generated")
  target_link_libraries(${example_target} PRIVATE Serializer::runtime)
  serializer_example_runtime_dlls(${example_target})
  if(MSVC)
    target_compile_options(${example_target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${example_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${example_target} COMMAND ${example_target}
    "${CMAKE_CURRENT_LIST_DIR}/../order.xml" "${CMAKE_CURRENT_BINARY_DIR}/${profile}/order.xml")
endforeach()
