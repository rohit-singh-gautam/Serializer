cmake_minimum_required(VERSION 4.0)
project(serializer_compression_exchange LANGUAGES CXX)
find_package(Serializer 2.14 CONFIG REQUIRED COMPONENTS runtime)
if(NOT SERIALIZER_GENERATOR_EXECUTABLE)
  message(FATAL_ERROR "SERIALIZER_GENERATOR_EXECUTABLE must select the compiler under qualification")
endif()
add_executable(compression_exchange_cpp compression_exchange.cpp)
serializer_generate(TARGET compression_exchange_cpp SCHEMAS model.serializer
  CONFIG model.ini GENERATOR "${SERIALIZER_GENERATOR_EXECUTABLE}")
if(MSVC)
  target_compile_options(compression_exchange_cpp PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(compression_exchange_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
