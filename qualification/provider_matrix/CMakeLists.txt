cmake_minimum_required(VERSION 4.0)
project(serializer_provider_matrix LANGUAGES CXX)

if(NOT EXISTS "${SERIALIZER_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Select the actual Serializer source checkout")
endif()
set(SERIALIZER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SERIALIZER_BUILD_MANAGED OFF CACHE BOOL "" FORCE)
set(SERIALIZER_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SERIALIZER_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SERIALIZER_SOURCE}" serializer EXCLUDE_FROM_ALL)
foreach(provider IN ITEMS ZSTD LZ4 ZLIB XML)
  if(SERIALIZER_WITH_${provider})
    set(provider_${provider} true)
  else()
    set(provider_${provider} false)
  endif()
endforeach()
configure_file(provider_expectations.hpp.in provider_expectations.hpp @ONLY)
add_executable(provider_matrix_test provider_matrix_test.cpp)
target_include_directories(provider_matrix_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(provider_matrix_test PRIVATE Serializer::runtime)
target_compile_features(provider_matrix_test PRIVATE cxx_std_23)
if(MSVC)
  target_compile_options(provider_matrix_test PRIVATE /W4 /WX)
else()
  target_compile_options(provider_matrix_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
