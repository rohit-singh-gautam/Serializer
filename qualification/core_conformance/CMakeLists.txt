cmake_minimum_required(VERSION 4.0)
project(serializer_core_conformance LANGUAGES CXX)

find_package(Serializer 2.21 CONFIG REQUIRED)
set(SERIALIZER_CORE_GENERATED_DIRECTORY "" CACHE PATH "Core C++ output from core_conformance.py --phase generate")
if(NOT EXISTS "${SERIALIZER_CORE_GENERATED_DIRECTORY}/core.hpp")
  message(FATAL_ERROR "SERIALIZER_CORE_GENERATED_DIRECTORY must contain the freshly generated Core header")
endif()
add_executable(core_conformance_probe ../core_conformance_probe.cpp)
target_include_directories(core_conformance_probe PRIVATE "${SERIALIZER_CORE_GENERATED_DIRECTORY}")
target_link_libraries(core_conformance_probe PRIVATE Serializer::runtime)
target_compile_features(core_conformance_probe PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(core_conformance_probe PRIVATE /W4 /WX)
else()
  target_compile_options(core_conformance_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
