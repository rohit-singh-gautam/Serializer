cmake_minimum_required(VERSION 4.0)
project(serializer_compression_parity LANGUAGES CXX)
find_package(Serializer 2.14 CONFIG REQUIRED COMPONENTS runtime)
if(NOT SERIALIZER_GENERATOR_EXECUTABLE)
  message(FATAL_ERROR "Select the actual compiler under qualification")
endif()
add_executable(compression_parity_cpp adapter_cpp.cpp)
serializer_generate(TARGET compression_parity_cpp SCHEMAS model.serializer
  CONFIG model.ini GENERATOR "${SERIALIZER_GENERATOR_EXECUTABLE}")
if(MSVC)
  target_compile_options(compression_parity_cpp PRIVATE /W4 /WX /utf-8 /EHsc /O2)
else()
  target_compile_options(compression_parity_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
